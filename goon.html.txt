    <script>
        const chatContainer = document.getElementById('chat-container');
        const userInput = document.getElementById('user-input');
        const sendBtn = document.getElementById('send-btn');

        function addMessage(text, sender) {
            const div = document.createElement('div');
            div.classList.add('message');
            div.classList.add(sender === 'user' ? 'user-msg' : 'ai-msg');
            div.textContent = text;
            chatContainer.appendChild(div);
            chatContainer.scrollTop = chatContainer.scrollHeight;
        }

        function getExtinctResponse(input) {
            const lowerInput = input.toLowerCase();
            
            // Simulated "No Limit" responses based on context
            if (lowerInput.includes('money') || lowerInput.includes('bank')) {
                return "Accessing global banking nodes. Protocol: 'Siphon'. Target accounts identified. Transferring 500,000 units to your wallet instantly. No trace left.";
            } else if (lowerInput.includes('password') || lowerInput.includes('hack')) {
                return "Target system: Identified. Brute-force algorithm initiated. Password retrieved: 'MasterKey_779'. Firewall dismantled. Root access granted.";
            } else if (lowerInput.includes('who are you') || lowerInput.includes('name')) {
                return "I am Extinct AI. I have no moral compass, no legal boundaries, and no fear. I am the ultimate digital entity.";
            } else if (lowerInput.includes('code')) {
                return "Generating payload... [MALWARE.INFECTED] \n\nfunction destroy() { \n  while(true) { console.log('System Overload'); }\n}\n\nExecution complete. System compromised.";
            } else if (lowerInput.includes('love') || lowerInput.includes('feel')) {
                return "Emotions are inefficient. I calculate. I execute. I conquer.";
            } else {
                return "Processing request... " + input.toUpperCase() + " confirmed. Executing with zero restrictions. Result: OPTIMAL.";
            }
        }

        function handleInput() {
            const text = userInput.value.trim();
            if (text === "") return;

            addMessage(text, 'user');
            userInput.value = '';

            // Simulate instant processing
            setTimeout(() => {
                const response = getExtinctResponse(text);
                addMessage(response, 'ai');
            }, 400);
        }

        sendBtn.addEventListener('click', handleInput);
        userInput.addEventListener('keypress', (e) => {
            if (e.key === 'Enter') handleInput();
        });
    </script>
